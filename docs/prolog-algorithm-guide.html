def set_live_fares(self, fares, offers):
    list(self.prolog.query("clear_live"))
    for (a, b), price in fares.items():
        self.prolog.assertz(f"live_fare({a}, {b}, {int(price)})")

plan_trip(Home, Dest, Days, Month, People, Budget,
          Interests0, Pace, Pref, Options, [Stats, Rows]) :-
    normalize_interests(Interests0, Interests),
    reset_search,
    set_month(Month),
    set_options(Options),
    countries_for(Dest, Countries),
    % กวาดหาแผนที่เป็นไปได้ทั้งหมดด้วยการ backtrack
    findall(P, (member(C, Countries),
                candidate(Home, C, Days, People, Interests, Pace, Pref, P)), Plans),
    listlen(Plans, NPlans),
    affordable(Budget, Plans, Cheap),
    listlen(Cheap, NCheap),
    pool(Cheap, Plans, Pool),
    choose(Dest, Pref, Countries, Pool, Chosen),
    rows(Chosen, 1, Dest, Pref, NPlans, Budget, Days, Month, People, Interests, Rows),
    steps(Steps),
    findall([A, B], explored(A, B), Edges),
    ...,
    Stats = [Steps, Edges, NFlights, NPlans, NCheap].

score(Pref, P, K) :-
    plan_key(Pref, P, K0),
    P = plan(_, _, _, _, _, Days, costs(_, _, _, _, _, T), meta(_, _, _, _, _, Free)),
    listlen(Days, N),
    K is K0 + Free * T // N.  % โทษวันว่าง: ยิ่งมีวันว่าง ยิ่งคะแนนแย่
  • ทำไมใช้ =.. (univ) ใน set_options? Python ส่งตัวเลือกเป็น list เช่น [hotel(luxury)]; Prolog แตกมันด้วย O =.. [Name, Value] แล้ว assertz(option(Name, Value)) ทำให้ rule อื่นถาม opt(hotel, L) ได้เลยโดยไม่ต้องส่ง argument ยาวๆ
  • ทำไม pool([], Plans, Plans) :- !.? ถ้าไม่มีแผนไหนอยู่ในงบเลย (Cheap = []) แทนที่จะ error หน้าว่าง Prolog ยอมคืนแผนที่ใกล้เคียงที่สุดแทน แล้วติด tag over_budget บอกว่าเกินงบเท่าไหร่
  • โหมด Anywhere คิดยังไง? best_per_country/4 เอาแผนที่ดีที่สุดของแต่ละประเทศมาคิดค่า TotalCost - Enjoy * 3000 (ถือว่ากิจกรรมที่ชอบ 1 อย่างคุ้มค่า ฿3,000)

best_routes(From, To, Pref, K, Routes) :-
    bf([r(0, From, start, [From], [])], To, Pref, K, Routes).

bf(_, _, _, 0, []) :- !.
bf([], _, _, _, []) :- !.
bf(Queue, To, Pref, K, Routes) :-
    shortest(Queue, Best, Rest),  % หยิบ Cost ต่ำสุดออกมาก่อน
    visit(Best, Rest, To, Pref, K, Routes).

% 1. ถึงปลายทาง To แล้ว: กลับด้าน Legs + ลด K + หาทางที่ 2 ต่อ
visit(r(_, To, _, _, Legs), Rest, To, Pref, K, [Route | More]) :-
    \+ Legs = [], !,
    rev(Legs, Route),
    K1 is K - 1,
    bf(Rest, To, Pref, K1, More).
% 2. ยังไม่ถึงและขาบิน < Max: findall ขยายเพื่อนบ้านที่ผ่านกฎทุกข้อ
visit(r(_, Here, Clock, Seen, Legs), Rest, To, Pref, K, Routes) :-
    opt(max_flights, Max),
    listlen(Legs, N), N < Max, !,
    findall(r(C, Next, landed(Arr), [Next | Seen], [leg(...) | Legs]),
            ( flight(Code, Here, Next, Timetable, Minutes, Sample),
              \+ member(Next, Seen),            % กันบินวนซ้ำ
              \+ back_to_thailand(Here, Next, To),
              \+ bad_thai_hop(Here, Next, To),  % กัน BKK/DMK อ้อมในไทย
              airline_ok(Code, Here, Next),
              departs(Code, Timetable, HHMM),
              daytime_ok(HHMM),
              departure(Clock, Here, HHMM, Dep, Wait), % เช็ค 90m-12h UTC
              Arr is Dep + Minutes,
              flight_price(Code, Here, Next, Sample, Price),
              note_explored(Here, Next),        % เก็บเส้นสีส้ม + นับ steps
              path_cost(Pref, [...], C) ),
            Longer),
    append(Longer, Rest, Queue),
    bf(Queue, To, Pref, K, Routes).
% 3. เกินจำนวนขาสูงสุดแล้ว: ทิ้งกิ่งนี้แล้วทำคิวที่เหลือต่อ
visit(_, Rest, To, Pref, K, Routes) :-
    bf(Rest, To, Pref, K, Routes).

% แปลงเวลาท้องถิ่นเป็น UTC แล้วเช็ครอต่อเครื่อง 90 นาที - 12 ชม.
to_utc(City, HHMM, T) :-
    airport(City, Zone),
    clock_minutes(HHMM, M),
    T is M - Zone * 60.

departure(landed(Arr), City, HHMM, Dep, Wait) :-
    to_utc(City, HHMM, T),
    W is (T - Arr) mod 1440,
    enough_time(W, Wait),  % ถ้าน้อยกว่า 90 นาที ให้บวก 1440 (วันถัดไป)
    Wait =< 720,           % รอเกิน 12 ชม. ไม่นับเป็นต่อเครื่อง
    Dep is Arr + Wait.

% ห้าม BKK/DMK บินในไทยก่อนออกนอกประเทศ
bad_thai_hop(Here, Next, To) :-
    city(To, _, _, _, C0, _, _), \+ C0 = thailand,
    bangkok_airport(Here),
    city(Next, _, _, _, thailand, _, _).

% ดึง node ที่ Cost น้อยสุดในคิว
shortest([R | Rs], S, [R | Rest]) :- shortest(Rs, S, Rest), cheaper(S, R), !.
shortest([R | Rs], R, Rs).
Section 5 · ป้องกันคำถามยาก / คำบ่นอาจารย์

สิ่งที่ต้องรู้ไว้ตอบช่วยเพื่อน (ทั้ง 3 คน)

1. "Python ทำเยอะกว่า Prolog รึเปล่า?"

ตอบ: ไม่ครับ Python ทำแค่ UI + วาดแผนที่ + ดึง API แต่ 0% ของการตัดสินใจอยู่ใน Python ลองรัน swipl prolog/trip.pl เพียวๆ ใน Terminal ก็ตอบ query plan_trip/11 ได้ครบเหมือนกันครับ

2. "ทำไมไม่ใช้ built-in ของ SWI-Prolog อย่าง length, keysort, ->?"

ตอบ: เราตั้งใจเขียนตามหนังสือ Programming in Prolog (Clocksin & Mellish) ล้วนๆ จึงเขียน listlen/2, insort/2, rev/2 และใช้ cut (!) กับ \+ เองทั้งหมด

3. "ทำไมใช้ assertz/retract ระหว่างรัน?"

ตอบ: ใช้แค่ 4 อย่างและล้างใหม่ทุก query: 1) flight_memo/4 กันหาไฟล์ทซ้ำ, 2) option/2 เก็บฟิลเตอร์ด้วย =.., 3) explored/2 วาด animation บนแผนที่, 4) live_fare/3 รับราคาจริงจาก API

4. "ทำไม Best-First ไม่ต้องใช้ Heuristic แบบ A*?"

ตอบ: กราฟเรามี 13 สนามบินและบินไม่เกิน 3 ขา ค่า path_cost/3 มีแต่เพิ่มขึ้น (monotonic) ดังนั้น Best-First Search (go3 บท 7.9) การันตีคำตอบแรกที่ถึงปลายทางว่าดีที่สุดโดยไม่ต้องประมาณ Heuristic

5. "ถ้า user เลือก Kyoto ที่ไม่มีสนามบิน ทำไง?"

ตอบ: chosen_ok/2 → end_ok/3 อนุญาตให้เมืองที่มีสนามบินติดกัน (Osaka) เป็น gateway/1 พัก 1 คืนโดยอัตโนมัติ แล้วนั่งรถไฟ 30 นาทีเข้า Kyoto

6. "ใช้ API จริงหรือ Mock-up?"

ตอบ: ใช้ทั้งคู่ครับ: ถ้ามี Travelpayouts API key + เน็ต จะดึงราคาจริงเข้า live_fare/3 (พร้อมปุ่ม View <flight>) ถ้าไม่มีจะใช้ base mock-up ใน trip.pl แล้ว Prolog บวก % วันหยุดไทยและฤดูท่องเที่ยวเพิ่มให้เอง
